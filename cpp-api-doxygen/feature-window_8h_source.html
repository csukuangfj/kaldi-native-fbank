32inline int32_t RoundUpToNearestPowerOfTwo(int32_t n) {
33 // copied from kaldi/src/base/kaldi-math.cc
34 if (n <= 0) {
35 throw std::invalid_argument("n must be > 0");
36 }
37 n--;
38 n |= n >> 1;
39 n |= n >> 2;
40 n |= n >> 4;
41 n |= n >> 8;
42 n |= n >> 16;
43 return n + 1;
44}

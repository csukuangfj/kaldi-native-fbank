38inline bool LoadNpy(const std::string &path, MatrixD *out) {
39 std::ifstream is(path, std::ios::binary);
40 if (!is) {
41 return false;
42 }
43 char magic[6];
44 is.read(magic, 6);
45 if (std::string(magic, 6) != "\x93NUMPY") {
46 return false;
47 }
48 char ver[2];
49 is.read(ver, 2);
50 uint32_t hlen = 0;
51 if (ver[0] == 1) {
52 uint16_t h16 = 0;
53 is.read(reinterpret_cast<char *>(&h16), 2);
54 hlen = h16;
55 } else {
56 is.read(reinterpret_cast<char *>(&hlen), 4);
57 }
58 std::string header(hlen, '\0');
59 is.read(&header[0], hlen);
60
61 auto dp = header.find("'descr': '") + 10;
62 std::string descr = header.substr(dp, 3);
63 auto sp = header.find("'shape': (") + 10;
64 auto ep = header.find(')', sp);
65 std::string shape = header.substr(sp, ep - sp);
66 int32_t rows = 0;
67 int32_t cols = 0;
68 if (shape.find(',') != std::string::npos) {
69 std::string col_part = shape.substr(shape.find(',') + 1);
70 rows = std::stoi(shape.substr(0, shape.find(',')));
71 // 1-D arrays have shape "(N,)", i.e. an empty column part.
72 cols = col_part.find_first_not_of(" \t") == std::string::npos
73 ? 1
74 : std::stoi(col_part);
75 } else {
76 rows = std::stoi(shape);
77 cols = 1;
78 }
79 *out = MatrixD(rows, cols);
80 if (descr == "<f8") {
81 for (int32_t r = 0; r < rows; ++r) {
82 is.read(reinterpret_cast<char *>((*out).row(r).data()),
83 cols * sizeof(double));
84 }
85 } else if (descr == "<i2") {
86 // int16 (e.g. raw wav samples stored as .npy).
87 std::vector<int16_t> tmp(cols);
88 for (int32_t r = 0; r < rows; ++r) {
89 is.read(reinterpret_cast<char *>(tmp.data()), cols * sizeof(int16_t));
90 for (int32_t c = 0; c < cols; ++c) {
91 (*out)(r, c) = tmp[c];
92 }
93 }
94 } else {
95 std::vector<float> tmp(cols);
96 for (int32_t r = 0; r < rows; ++r) {
97 is.read(reinterpret_cast<char *>(tmp.data()), cols * sizeof(float));
98 for (int32_t c = 0; c < cols; ++c) {
99 (*out)(r, c) = tmp[c];
100 }
101 }
102 }
103 return true;
104}

120inline double MaxAbsDiff(const MatrixF &a, const MatrixD &b) {
121 double max_diff = 0.0;
122 for (int32_t r = 0; r < a.rows(); ++r) {
123 for (int32_t c = 0; c < a.cols(); ++c) {
124 double diff = std::fabs(static_cast<double>(a(r, c)) - b(r, c));
125 if (diff > max_diff) {
126 max_diff = diff;
127 }
128 }
129 }
130 return max_diff;
131}
